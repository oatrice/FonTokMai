### Manual Verification Guide: Cache Stampede & Coroutine Fix

**Objective:** Verify that `WeatherManager` handles concurrent requests without recalculating optical flow for the same station, and ensure that old coroutine/name errors are resolved.

- **Step 1:** เปิด Terminal แล้วไปที่โฟลเดอร์ `backend` จากนั้นสร้างไฟล์สคริปต์ชื่อ `test_concurrency.py` โดยก๊อปปี้โค้ดด้านล่างนี้ไปวาง:

```python
import asyncio
import logging
from unittest.mock import AsyncMock, patch

# เปิด Logs เพื่อสังเกตการทำงาน
logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")

async def main():
    from app.services.weather_manager import WeatherManager
    
    # เราจะ Mock ส่วนที่ต่อ Database และ Storage เพื่อให้รันได้โดยไม่ติดปัญหา Local DB Schema 
    # หรือปัญหาสิทธิการเข้าถึง Cloud Storage และเพื่อวัดจำนวนครั้งที่โหลดจริงๆ
    
    with patch("app.dependencies.get_repo_context") as mock_repo, \\
         patch("google.cloud.storage.Client"), \\
         patch("app.services.tmd_radar_processor.TMDRadarProcessor.calculate_optical_flow") as mock_flow:
        
        # สมมติให้ Optical Flow โหลดช้า 1 วินาที เพื่อจำลองการแย่งกันของ Thread
        async def slow_optical_flow(*args, **kwargs):
            await asyncio.sleep(1)
            return []
            
        mock_flow.side_effect = lambda *args: asyncio.run(slow_optical_flow())
        
        # Setup Mock DB
        mock_repo_instance = AsyncMock()
        mock_repo_instance.get_latest_radar_cache.return_value = {
            "url_t": "dummy_t",
            "url_t_minus_1": "dummy_t_minus_1",
            "timestamp": 1234567890
        }
        mock_repo.return_value.__aenter__.return_value = mock_repo_instance

        # Setup Mock Download
        with patch("asyncio.gather") as mock_gather:
            import numpy as np
            import cv2
            dummy_img = np.zeros((800, 800, 3), dtype=np.uint8)
            success, encoded = cv2.imencode('.png', dummy_img)
            mock_gather.return_value = [encoded.tobytes(), encoded.tobytes()]
            
            wm = WeatherManager()
            
            # พิกัดหนองคาย (17.8785, 102.7420) ที่อยู่ในรัศมีเรดาร์ skn240
            lat, lng = 17.8785, 102.7420
            
            print("🚀 ยิง 15 Requests พร้อมกัน (Concurrent) ไปที่ TMD Radar...")
            
            tasks = [wm._get_tmd_prediction(lat, lng) for _ in range(15)]
            results = await asyncio.gather(*tasks, return_exceptions=True)
            
            successes = sum(1 for r in results if not isinstance(r, Exception))
            errors = sum(1 for r in results if isinstance(r, Exception))
            
            print(f"\\n✅ ทำงานเสร็จสิ้น! สำเร็จ: {successes}, เออเร่อ: {errors}")
            print(f"🔍 จำนวนครั้งที่โหลดข้อมูลและคำนวณ Optical Flow จริง: {mock_flow.call_count} ครั้ง (ควรเป็น 1 ครั้ง)")

if __name__ == "__main__":
    asyncio.run(main())
```
